    <script>
        const SUPABASE_URL = 'https://lcwyynpazzkmdlsmmrrq.supabase.co';
        const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Imxjd3l5bnBhenprbWRsc21tcnJxIiwicm9sZSI6ImFub24iLCJpYXQiOjE5MjI3Mzg1Mjl9.B7m_W1YvQc9wLqS3k_v6pT8xZ2wF1v3Q4s5d6f7g8h9'; 
        
        const supabaseClient = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

        window.addEventListener('DOMContentLoaded', async () => {
            const { data: { session } } = await supabaseClient.auth.getSession();
            if (session) {
                await verifyAndLoadDashboard(session.user.id);
            }
        });

        async function handleLogin(e) {
            e.preventDefault();
            const email = document.getElementById('admin-email').value.trim();
            const password = document.getElementById('admin-password').value;
            const errorBox = document.getElementById('login-error');
            const loginBtn = document.getElementById('login-btn');

            errorBox.classList.add('hidden');
            loginBtn.disabled = true;
            loginBtn.innerText = 'Verifying...';

            try {
                const { data, error } = await supabaseClient.auth.signInWithPassword({ email, password });
                if (error) throw error;

                await verifyAndLoadDashboard(data.user.id);
            } catch (err) {
                errorBox.innerText = err.message || 'Invalid login credentials';
                errorBox.classList.remove('hidden');
                loginBtn.disabled = false;
                loginBtn.innerText = 'Secure Login';
            }
        }

        async function verifyAndLoadDashboard(userId) {
            const errorBox = document.getElementById('login-error');
            const loginBtn = document.getElementById('login-btn');

            try {
                const { data, error } = await supabaseClient
                    .from('admin_users')
                    .select('*')
                    .eq('user_id', userId)
                    .maybeSingle();

                if (error || !data) {
                    await supabaseClient.auth.signOut();
                    throw new Error('Access Denied: User is not authorized as Administrator.');
                }

                document.getElementById('login-section').classList.add('hidden');
                document.getElementById('dashboard-section').classList.remove('hidden');
                loadDashboardStats();
            } catch (err) {
                errorBox.innerText = err.message;
                errorBox.classList.remove('hidden');
                if(loginBtn) {
                    loginBtn.disabled = false;
                    loginBtn.innerText = 'Secure Login';
                }
            }
        }

        async function loadDashboardStats() {
            try {
                const { count: bookingCount } = await supabaseClient.from('bookings').select('*', { count: 'exact', head: true });
                document.getElementById('stat-bookings').innerText = bookingCount !== null ? bookingCount : '0';

                const { count: productCount } = await supabaseClient.from('jewellery_inventory').select('*', { count: 'exact', head: true });
                document.getElementById('stat-products').innerText = productCount !== null ? productCount : '0';
            } catch (e) {
                console.error('Stats load error:', e);
            }
        }

        async function handleLogout() {
            await supabaseClient.auth.signOut();
            document.getElementById('dashboard-section').classList.add('hidden');
            document.getElementById('login-section').classList.remove('hidden');
            document.getElementById('login-form').reset();
            const loginBtn = document.getElementById('login-btn');
            loginBtn.disabled = false;
            loginBtn.innerText = 'Secure Login';
        }
    </script>
